local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local RunService = game:GetService("RunService")
local UserInputService = game:GetService("UserInputService")

local LocalPlayer = Players.LocalPlayer
local Config = { FlySpeed = 80, WalkSpeed = 16, Noclip = false, Fly = false, GodMode = false, AutoRevive = true }

local function load()
    local Character = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
    local RootPart = Character:WaitForChild("HumanoidRootPart")
    local Humanoid = Character:WaitForChild("Humanoid")

    -- NOCLIP E VOO
    RunService.RenderStepped:Connect(function()
        if not Character then return end
        if Config.Noclip then
            for _, part in pairs(Character:GetChildren()) do
                if part:IsA("BasePart") and part.CanCollide == true then part.CanCollide = false end
            end
        else
            for _, part in pairs(Character:GetChildren()) do
                if part:IsA("BasePart") and part.ClassName ~= "Motor6D" then part.CanCollide = true end
            end
        end
        
        if Config.Fly then
            local MoveDir = Humanoid.MoveDirection
            local CamCFrame = workspace.CurrentCamera.CFrame
            local Fwd = CamCFrame.RightVector:Dot(MoveDir)
            local Rgt = CamCFrame.LookVector:Dot(MMoveDir)
            local Up = 0
            if UserInputService:IsKeyDown(Enum.KeyCode.W) then Up = 1 end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then Up = -1 end
            RootPart.Velocity = Vector3.new(Fwd * Config.FlySpeed, Up * Config.FlySpeed, Rgt * Config.FlySpeed)
            Humanoid.WalkSpeed = 16
        else
            Humanoid.WalkSpeed = Config.WalkSpeed
        end
    end)

    -- GOD MODE
    RunService.Heartbeat:Connect(function()
        if Config.GodMode and Humanoid.Health < Humanoid.MaxHealth then
            Humanoid.Health = Humanoid.MaxHealth
        end
    end)

    -- AUTO REVIVE SINTONIA RP
    RunService.Heartbeat:Connect(function()
        if not Config.AutoRevive or Character.Humanoid.Health > 0 then return end
        task.spawn(function()
            task.wait(1)
            local OrpheusShared = ReplicatedStorage:FindFirstChild("OrpheusShared")
            if OrpheusShared then
                local DeathEvent = OrpheusShared:FindFirstChild("Death")
                if DeathEvent and DeathEvent:IsA("RemoteEvent") then
                    DeathEvent:FireServer("Action1")
                end
            end
        end)
    end)

    -- INTERFACE
    local ScreenGui = Instance.new("ScreenGui")
    ScreenGui.Name = "JCHubUI"
    ScreenGui.ZIndexBehavior = Enum.ZIndexBehavior.Foreground
    local MainFrame = Instance.new("Frame")
    MainFrame.Size = UDim2.new(0, 350, 0, 350)
    MainFrame.Position = UDim2.new(0.5, -175, 0.5, -175)
    MainFrame.BackgroundColor3 = Color3.fromRGB(20, 25, 35)
    MainFrame.BackgroundTransparency = 0.1
    MainFrame.BorderSizePixel = 0
    MainFrame.Parent = ScreenGui
    local Corner = Instance.new("UICorner"); Corner.CornerRadius = UDim.new(0, 10); Corner.Parent = MainFrame
    
    local TitleBar = Instance.new("Frame")
    TitleBar.Size = UDim2.new(1, 0, 0, 40)
    TitleBar.BackgroundColor3 = Color3.fromRGB(0, 100, 255)
    TitleBar.BorderSizePixel = 0; TitleBar.Parent = MainFrame
    local TitleText = Instance.new("TextLabel"); TitleText.Size = UDim2.new(1, -20, 1, 0); TitleText.Text = "JC HUB"; TitleText.Font = Enum.Font.GothamBold; TitleText.TextColor3 = Color3.fromRGB(255, 255, 255); TitleText.Parent = TitleBar
    
    local CloseBtn = Instance.new("TextButton"); CloseBtn.Size = UDim2.new(0, 30, 0, 30); CloseBtn.Text = "X"; CloseBtn.Parent = TitleBar
    CloseBtn.MouseButton1Click:Connect(function() ScreenGui:Destroy() end)

    local Container = Instance.new("ScrollingFrame")
    Container.Size = UDim2.new(1, -20, 1, -60)
    Container.BackgroundTransparency = 1; Container.ScrollBarThickness = 4; Container.Parent = MainFrame
    local ListLayout = Instance.new("UIListLayout"); ListLayout.Padding = UDim.new(0, 5); ListLayout.Parent = Container

    local function btn(text, func)
        local b = Instance.new("TextButton")
        b.Size = UDim2.new(1, 0, 0, 35)
        b.BackgroundColor3 = Color3.fromRGB(30, 40, 60)
        b.Text = text; b.TextColor3 = Color3.fromRGB(200, 230, 255); b.Font = Enum.Font.GothamSemibold
        b.Parent = Container
        local c = Instance.new("UICorner"); c.CornerRadius = UDim.new(0, 5); c.Parent = b
        b.MouseButton1Click:Connect(func)
    end

    btn("Super Speed", function() Config.WalkSpeed = 45 end)
    btn("Toggle Voo", function() Config.Fly = not Config.Fly end)
    btn("Noclip", function() Config.Noclip = not Config.Noclip end)
    btn("God Mode", function() Config.GodMode = not Config.GodMode end)
    btn("Auto Revive", function() Config.AutoRevive = not Config.AutoRevive end)
    btn("Resetar", function() Config.Fly = false; Config.Noclip = false; Config.WalkSpeed = 16 end)

    ScreenGui.Parent = LocalPlayer.PlayerGui
end

load()
LocalPlayer.CharacterAdded:Connect(load)
